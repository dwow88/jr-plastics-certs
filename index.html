#!/usr/bin/env python3 """ JR Plastics Certificate Processor - Professional Desktop App Matches the website UI/UX with modern design """ import tkinter as tk from tkinter import ttk, filedialog, messagebox, scrolledtext import os import json import base64 import time import requests from pathlib import Path import threading import fitz # PyMuPDF for PDF handling class JRPlasticsProApp: def __init__(self, root): self.root = root self.root.title("JR Plastics Certificate Processor") self.root.geometry("1100x800") self.root.minsize(1000, 700) # Color scheme matching website self.bg_main = "#f5f0e8" self.bg_header = "#1a1410" self.text_light = "#e8d5b0" self.accent_green = "#2a6a3e" self.accent_light_green = "#40c070" self.error_red = "#c0392b" self.text_dark = "#2c2c2c" self.root.configure(bg=self.bg_main) # Data storage self.api_key = "" self.pdf_path = "" self.results = [] self.config_file = Path.home() / ".jr_plastics_config.json" self.is_processing = False # Load saved API key self.load_config() # Build beautiful UI self.build_ui() def load_config(self): """Load saved API key""" if self.config_file.exists(): try: with open(self.config_file, 'r') as f: config = json.load(f) self.api_key = config.get('api_key', '') except: pass def build_ui(self): """Build the professional UI""" # Header with logo area header = tk.Frame(self.root, bg=self.bg_header, height=80) header.pack(fill=tk.X) header.pack_propagate(False) # Title title_frame = tk.Frame(header, bg=self.bg_header) title_frame.pack(side=tk.LEFT, fill=tk.BOTH, expand=True, padx=20, pady=15) title = tk.Label(title_frame, text="JR Plastics", font=("Arial", 24, "bold"), fg=self.text_light, bg=self.bg_header) title.pack(anchor=tk.W) subtitle = tk.Label(title_frame, text="Certificate Processor", font=("Arial", 10), fg="#a0a0a0", bg=self.bg_header) subtitle.pack(anchor=tk.W) # Main container with padding main_container = tk.Frame(self.root, bg=self.bg_main) main_container.pack(fill=tk.BOTH, expand=True, padx=20, pady=20) # Create two columns left_column = tk.Frame(main_container, bg=self.bg_main) left_column.pack(side=tk.LEFT, fill=tk.BOTH, expand=True, padx=(0, 10)) right_column = tk.Frame(main_container, bg=self.bg_main) right_column.pack(side=tk.RIGHT, fill=tk.BOTH, expand=True, padx=(10, 0)) # --- LEFT COLUMN --- # API Key Section api_frame = self.create_card(left_column, "Groq API Key") info_label = tk.Label(api_frame, text="Get key at: https://console.groq.com/keys", font=("Arial", 9), fg="#666666", bg="white") info_label.pack(anchor=tk.W, padx=15, pady=(10, 10)) key_input_frame = tk.Frame(api_frame, bg="white") key_input_frame.pack(fill=tk.X, padx=15, pady=(0, 15)) self.key_entry = tk.Entry(key_input_frame, font=("Arial", 10), show="*", bg="#f9f9f9", fg=self.text_dark, relief=tk.FLAT, bd=0) self.key_entry.pack(fill=tk.X, ipady=8) self.key_entry.insert(0, self.api_key) # Separator sep = tk.Frame(key_input_frame, bg="#e0e0e0", height=1) sep.pack(fill=tk.X, pady=5) save_btn = tk.Button(key_input_frame, text="Save API Key", command=self.save_api_key, font=("Arial", 9, "bold"), bg=self.accent_green, fg="white", relief=tk.FLAT, bd=0, padx=15, pady=8, cursor="hand2", activebackground=self.accent_light_green) save_btn.pack(anchor=tk.W) # File Selection Section file_frame = self.create_card(left_column, "Select PDF File") file_btn = tk.Button(file_frame, text="Choose PDF File", command=self.choose_file, font=("Arial", 10, "bold"), bg=self.accent_green, fg="white", relief=tk.FLAT, bd=0, padx=20, pady=10, cursor="hand2", activebackground=self.accent_light_green) file_btn.pack(pady=10) self.file_label = tk.Label(file_frame, text="No file selected", font=("Arial", 9), fg="#888888", bg="white") self.file_label.pack(anchor=tk.W, padx=15, pady=(0, 15)) # Processing Section process_frame = self.create_card(left_column, "Process") self.process_btn = tk.Button(process_frame, text="Process All Pages", command=self.process_pdf, font=("Arial", 11, "bold"), bg=self.accent_green, fg="white", relief=tk.FLAT, bd=0, padx=20, pady=12, cursor="hand2", activebackground=self.accent_light_green) self.process_btn.pack(pady=10) # Progress self.progress_var = tk.StringVar(value="Ready to process") self.progress_label = tk.Label(process_frame, textvariable=self.progress_var, font=("Arial", 9), fg="#666666", bg="white") self.progress_label.pack(anchor=tk.W, padx=15, pady=(0, 10)) self.progress_bar = ttk.Progressbar(process_frame, mode='determinate', length=300) self.progress_bar.pack(fill=tk.X, padx=15, pady=(0, 15)) # Export buttons export_frame = tk.Frame(process_frame, bg="white") export_frame.pack(fill=tk.X, padx=15, pady=(0, 15)) csv_btn = tk.Button(export_frame, text="Export CSV", command=self.export_csv, font=("Arial", 9), bg="#f0f0f0", fg=self.text_dark, relief=tk.FLAT, bd=0, padx=12, pady=8, cursor="hand2") csv_btn.pack(side=tk.LEFT, padx=(0, 5)) json_btn = tk.Button(export_frame, text="Export JSON", command=self.export_json, font=("Arial", 9), bg="#f0f0f0", fg=self.text_dark, relief=tk.FLAT, bd=0, padx=12, pady=8, cursor="hand2") json_btn.pack(side=tk.LEFT) # --- RIGHT COLUMN --- # Results Section results_frame = self.create_card(right_column, "Results Preview", height=400) self.results_text = scrolledtext.ScrolledText(results_frame, height=20, width=50, font=("Courier", 9), bg="#f9f9f9", fg=self.text_dark, relief=tk.FLAT, bd=0) self.results_text.pack(fill=tk.BOTH, expand=True, padx=15, pady=15) self.results_text.config(state=tk.DISABLED) # Status bar status_frame = tk.Frame(self.root, bg="#2c2c2c", height=30) status_frame.pack(fill=tk.X, side=tk.BOTTOM) status_frame.pack_propagate(False) self.status_var = tk.StringVar(value="Ready") status_label = tk.Label(status_frame, textvariable=self.status_var, font=("Arial", 9), fg="#e0e0e0", bg="#2c2c2c") status_label.pack(anchor=tk.W, padx=15, pady=5) def create_card(self, parent, title, height=None): """Create a styled card frame""" card = tk.Frame(parent, bg="white", relief=tk.FLAT, bd=1) card.pack(fill=tk.X, pady=(0, 15)) if height: card.pack(fill=tk.BOTH, expand=True) # Card header header = tk.Frame(card, bg=self.bg_header) header.pack(fill=tk.X) title_label = tk.Label(header, text=title, font=("Arial", 11, "bold"), fg=self.text_light, bg=self.bg_header) title_label.pack(anchor=tk.W, padx=15, pady=10) return card def save_api_key(self): """Save API key""" self.api_key = self.key_entry.get().strip() if not self.api_key: messagebox.showerror("Error", "Please enter your Groq API key") return try: with open(self.config_file, 'w') as f: json.dump({'api_key': self.api_key}, f) messagebox.showinfo("Success", "API key saved securely") self.status_var.set("API key saved") except Exception as e: messagebox.showerror("Error", f"Failed to save: {e}") def choose_file(self): """Choose PDF file""" file_path = filedialog.askopenfilename( title="Select PDF File", filetypes=[("PDF files", "*.pdf"), ("All files", "*.*")] ) if file_path: self.pdf_path = file_path filename = Path(file_path).name self.file_label.config(text=f"Selected: {filename}", fg=self.text_dark) self.status_var.set(f"Loaded: {filename}") def pdf_to_images(self, pdf_path): """Convert PDF pages to base64 JPEG images""" images = [] try: pdf = fitz.open(pdf_path) for page_num in range(len(pdf)): page = pdf[page_num] pix = page.get_pixmap(matrix=fitz.Matrix(2, 2)) image_data = pix.tobytes("jpeg") b64 = base64.b64encode(image_data).decode('utf-8') images.append(b64) pdf.close() return images except Exception as e: raise Exception(f"Failed to convert PDF: {e}") def extract_data(self, image_base64): """Call Groq API to extract data""" if not self.api_key: raise Exception("No Groq API key configured") prompt = """CRITICAL: Extract hand-written thickness gauges from JR Plastics certification form. GAUGE LAYOUT: - TOP ROW: 9 gauges (t1-t9), read LEFT TO RIGHT - BOTTOM ROW: 10 gauges (t10-t19), read LEFT TO RIGHT INSTRUCTIONS: 1. Carefully read EACH hand-written number 2. Gauges are decimal measurements (e.g., 0.125, 0.250, 1.500) 3. Read numbers even if slightly smudged or unclear 4. If unreadable, use empty string "" 5. Do NOT guess or estimate Return ONLY valid JSON with ALL fields: {"po_number":"", "pallet_number":"", "railcar_number":"", "date":"", "pieces":"", "packer":"", "operator":"", "t1":"", "t2":"", "t3":"", "t4":"", "t5":"", "t6":"", "t7":"", "t8":"", "t9":"", "t10":"", "t11":"", "t12":"", "t13":"", "t14":"", "t15":"", "t16":"", "t17":"", "t18":"", "t19":"", "length":"", "width":"", "surface_finish":"", "qc_date":"", "upf_receiver":"", "recv_date":"", "recv_time":""}""" headers = { 'Content-Type': 'application/json', 'Authorization': f'Bearer {self.api_key}' } payload = { 'model': 'qwen/qwen3.6-27b', 'max_tokens': 2000, 'messages': [{ 'role': 'user', 'content': [ { 'type': 'image_url', 'image_url': {'url': f'data:image/jpeg;base64,{image_base64}'} }, {'type': 'text', 'text': prompt} ] }] } for attempt in range(5): # More retries for rate limiting try: response = requests.post( 'https://api.groq.com/openai/v1/chat/completions', headers=headers, json=payload, timeout=30 ) if response.status_code == 429: if attempt < 4: wait_time = 15 + (attempt * 5) # Exponential backoff: 15, 20, 25, 30 seconds time.sleep(wait_time) continue raise Exception("Rate limit exceeded after retries - upgrade Groq account") if not response.ok: raise Exception(f"API error {response.status_code}") data = response.json() text = data.get('choices', [{}])[0].get('message', {}).get('content', '') if not text: raise Exception("Empty response") # Clean response text = text.replace('', '').replace('', '').strip() text = text.replace('```json', '').replace('```', '').strip() # Try to parse JSON try: return json.loads(text) except: # Try to extract JSON object import re match = re.search(r'\{[\s\S]*\}', text) if match: return json.loads(match.group()) raise Exception("Could not parse JSON") except Exception as e: if attempt == 4: raise if "rate limit" in str(e).lower(): wait_time = 15 + (attempt * 5) time.sleep(wait_time) else: time.sleep(3) def process_pdf(self): """Process PDF in background""" if not self.pdf_path: messagebox.showerror("Error", "Please select a PDF file first") return if not self.api_key: messagebox.showerror("Error", "Please save your Groq API key first") return if self.is_processing: messagebox.showwarning("Warning", "Already processing a file") return self.is_processing = True self.process_btn.config(state=tk.DISABLED) thread = threading.Thread(target=self._process_pdf_thread) thread.start() def _process_pdf_thread(self): """Background processing thread""" try: self.progress_var.set("Converting PDF to images...") self.root.update() images = self.pdf_to_images(self.pdf_path) self.results = [] self.progress_bar.config(maximum=len(images)) if len(images) > 1: self.status_var.set(f"Processing {len(images)} pages (this may take a few minutes due to Groq rate limits)...") for i, img_b64 in enumerate(images): if not self.is_processing: break self.progress_var.set(f"Processing page {i+1} of {len(images)}...") self.progress_bar.config(value=i) self.root.update() try: extracted = self.extract_data(img_b64) extracted['page'] = i + 1 self.results.append(extracted) except Exception as e: self.results.append({ 'page': i + 1, 'error': str(e) }) if i < len(images) - 1: time.sleep(10) # 10 second delay between pages to avoid rate limiting self.progress_bar.config(value=len(images)) self.progress_var.set(f"Complete! Processed {len(self.results)} pages") self.display_results() self.status_var.set(f"Successfully processed {len(images)} pages") messagebox.showinfo("Success", f"Processed {len(images)} pages") except Exception as e: self.progress_var.set("Error occurred") self.status_var.set(f"Error: {str(e)}") messagebox.showerror("Error", f"Processing failed: {str(e)}") finally: self.is_processing = False self.process_btn.config(state=tk.NORMAL) def display_results(self): """Display results and show gauge verification if available""" self.results_text.config(state=tk.NORMAL) self.results_text.delete(1.0, tk.END) # Check if we have gauges to verify has_gauges = False for result in self.results: if 'error' not in result: for i in range(1, 20): if f't{i}' in result: has_gauges = True break # Show results for result in self.results: self.results_text.insert(tk.END, f"=== Page {result.get('page', '?')} ===\n") if 'error' in result: self.results_text.insert(tk.END, f"ERROR: {result['error']}\n") else: for key, value in sorted(result.items()): if key != 'page': self.results_text.insert(tk.END, f"{key}: {value}\n") self.results_text.insert(tk.END, "\n") self.results_text.config(state=tk.DISABLED) # Show gauge verification dialog if we have gauges if has_gauges and len(self.results) > 0: self.root.after(500, self.show_gauge_verification) def show_gauge_verification(self): """Show dialog to verify and edit gauge readings""" if not self.results or 'error' in self.results[-1]: return last_result = self.results[-1] # Create verification window verify_win = tk.Toplevel(self.root) verify_win.title("Verify Gauge Readings") verify_win.geometry("700x500") verify_win.resizable(False, False) # Header header = tk.Label(verify_win, text="Review Hand-Written Gauge Readings", font=("Arial", 12, "bold"), bg=self.bg_header, fg=self.text_light) header.pack(fill=tk.X, padx=0, pady=0) # Instructions instr = tk.Label(verify_win, text="Edit any incorrect readings. Top row: 9 gauges | Bottom row: 10 gauges", font=("Arial", 9), fg="#666666") instr.pack(pady=10) # Gauge entries frame gauge_frame = tk.Frame(verify_win) gauge_frame.pack(fill=tk.BOTH, expand=True, padx=15, pady=10) gauge_entries = {} # Top row (9 gauges) top_row = tk.Frame(gauge_frame) top_row.pack(fill=tk.X, pady=(0, 15)) tk.Label(top_row, text="Top Row (t1-t9):", font=("Arial", 9, "bold")).pack(anchor=tk.W) top_grid = tk.Frame(top_row) top_grid.pack(fill=tk.X) for i in range(1, 10): col = tk.Frame(top_grid) col.pack(side=tk.LEFT, padx=5) tk.Label(col, text=f"t{i}", font=("Arial", 8, "bold")).pack() entry = tk.Entry(col, width=8, font=("Arial", 9)) entry.pack() entry.insert(0, str(last_result.get(f't{i}', ''))) gauge_entries[f't{i}'] = entry # Bottom row (10 gauges) bottom_row = tk.Frame(gauge_frame) bottom_row.pack(fill=tk.X) tk.Label(bottom_row, text="Bottom Row (t10-t19):", font=("Arial", 9, "bold")).pack(anchor=tk.W) bottom_grid = tk.Frame(bottom_row) bottom_grid.pack(fill=tk.X) for i in range(10, 20): col = tk.Frame(bottom_grid) col.pack(side=tk.LEFT, padx=5) tk.Label(col, text=f"t{i}", font=("Arial", 8, "bold")).pack() entry = tk.Entry(col, width=8, font=("Arial", 9)) entry.pack() entry.insert(0, str(last_result.get(f't{i}', ''))) gauge_entries[f't{i}'] = entry # Buttons button_frame = tk.Frame(verify_win) button_frame.pack(fill=tk.X, padx=15, pady=15) def save_gauges(): """Save corrected gauge values""" for key, entry in gauge_entries.items(): last_result[key] = entry.get().strip() # Update display self.display_results() verify_win.destroy() messagebox.showinfo("Success", "Gauge readings saved!") save_btn = tk.Button(button_frame, text="Save Corrections", command=save_gauges, font=("Arial", 10, "bold"), bg=self.accent_green, fg="white", relief=tk.FLAT, bd=0, padx=20, pady=8, cursor="hand2") save_btn.pack(side=tk.LEFT, padx=(0, 10)) cancel_btn = tk.Button(button_frame, text="Cancel", command=verify_win.destroy, font=("Arial", 10), bg="#f0f0f0", fg=self.text_dark, relief=tk.FLAT, bd=0, padx=20, pady=8, cursor="hand2") cancel_btn.pack(side=tk.LEFT) """Export to CSV""" if not self.results: messagebox.showerror("Error", "No results to export") return file_path = filedialog.asksaveasfilename( defaultextension=".csv", filetypes=[("CSV files", "*.csv")] ) if not file_path: return try: import csv with open(file_path, 'w', newline='') as f: if self.results: all_keys = set() for result in self.results: all_keys.update(result.keys()) all_keys = sorted(list(all_keys)) writer = csv.DictWriter(f, fieldnames=all_keys) writer.writeheader() writer.writerows(self.results) messagebox.showinfo("Success", f"Exported to {Path(file_path).name}") self.status_var.set("Exported to CSV") except Exception as e: messagebox.showerror("Error", f"Export failed: {e}") def export_json(self): """Export to JSON""" if not self.results: messagebox.showerror("Error", "No results to export") return file_path = filedialog.asksaveasfilename( defaultextension=".json", filetypes=[("JSON files", "*.json")] ) if not file_path: return try: with open(file_path, 'w') as f: json.dump(self.results, f, indent=2) messagebox.showinfo("Success", f"Exported to {Path(file_path).name}") self.status_var.set("Exported to JSON") except Exception as e: messagebox.showerror("Error", f"Export failed: {e}") if __name__ == "__main__": root = tk.Tk() app = JRPlasticsProApp(root) root.mainloop()
